import pygame import sys import os # 1. Inicialização do Pygame e do Mixer de Áudio pygame.init() pygame.mixer.init() # Configurações da Tela LARGURA, ALTURA = 800, 600 tela = pygame.display.set_mode((LARGURA, ALTURA)) pygame.display.set_caption("Jogo Corrigido") relogio = pygame.time.Clock() # 2. Carregamento do Plano de Fundo (com Fallback se o arquivo não existir) ARQUIVO_FUNDO = "fundo.png" if os.path.exists(ARQUIVO_FUNDO): imagem_fundo = pygame.image.load(ARQUIVO_FUNDO).convert() imagem_fundo = pygame.transform.scale(imagem_fundo, (LARGURA, ALTURA)) else: # Se a imagem sumir ou não existir, cria um fundo azul de emergência imagem_fundo = pygame.Surface((LARGURA, ALTURA)) imagem_fundo.fill((30, 30, 60)) # 3. Carregamento do Personagem (com Fallback) ARQUIVO_PERSONAGEM = "personagem.png" if os.path.exists(ARQUIVO_PERSONAGEM): imagem_personagem = pygame.image.load(ARQUIVO_PERSONAGEM).convert_alpha() imagem_personagem = pygame.transform.scale(imagem_personagem, (50, 50)) else: # Se a imagem sumir, cria um quadrado vermelho de emergência imagem_personagem = pygame.Surface((50, 50)) imagem_personagem.fill((255, 50, 50)) # Variáveis de Posição do Personagem pos_x, pos_y = 375, 275 velocidade = 5 # 4. Carregamento e Reprodução de Áudio ARQUIVO_AUDIO = "musica.mp3" # pode ser .wav ou .ogg if os.path.exists(ARQUIVO_AUDIO): pygame.mixer.music.load(ARQUIVO_AUDIO) pygame.mixer.music.play(-1) # -1 faz o áudio tocar em loop infinito pygame.mixer.music.set_volume(0.5) # 5. Loop Principal do Jogo rodando = True while rodando: relogio.tick(60) # Mantém 60 FPS fixos # Captura de Eventos for evento in pygame.event.get(): if evento.type == pygame.QUIT: rodando = False # Controles do Personagem teclas = pygame.key.get_pressed() if teclas[pygame.K_LEFT] or teclas[pygame.K_a]: pos_x -= velocidade if teclas[pygame.K_RIGHT] or teclas[pygame.K_d]: pos_x += velocidade if teclas[pygame.K_UP] or teclas[pygame.K_w]: pos_y -= velocidade if teclas[pygame.K_DOWN] or teclas[pygame.K_s]: pos_y += velocidade # Limites da Tela (impede o personagem de sumir saindo da janela) pos_x = max(0, min(pos_x, LARGURA - 50)) pos_y = max(0, min(pos_y, ALTURA - 50)) # --- DESENHO NA TELA (Ordem Crítica) --- # 1º Desenha o Fundo tela.blit(imagem_fundo, (0, 0)) # 2º Desenha o Personagem por cima do fundo tela.blit(imagem_personagem, (pos_x, pos_y)) # 3º Atualiza a Tela pygame.display.flip() pygame.quit() sys.exit()
